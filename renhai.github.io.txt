    <script>
        // 核心变量：选中的按钮集合（无顺序）
        let selectedButtons = new Set();
        let initA = 120;
        let initB = 120;
        let targetA = 60;
        let targetB = 210;

        const resultA = document.getElementById('resultA');
        const resultB = document.getElementById('resultB');
        const triggerBtns = document.querySelectorAll('.trigger-btn');
        const resetBtn = document.getElementById('reset');
        const autoSolveBtn = document.getElementById('autoSolve');
        const clearInputsBtn = document.getElementById('clearInputs'); // 清空数字按钮
        const solutionTip = document.getElementById('solutionTip');

        // 初始化数值
        function initValues() {
            initA = Number(document.getElementById('initA').value) || 0;
            initB = Number(document.getElementById('initB').value) || 0;
            targetA = Number(document.getElementById('targetA').value) || 0;
            targetB = Number(document.getElementById('targetB').value) || 0;
            resultA.textContent = initA;
            resultB.textContent = initB;
        }

        // 解析单个按钮的运算规则
        function parseOp(inputValue) {
            if (!inputValue || inputValue.trim() === '' || inputValue.trim() === '0') {
                return { op: '', num: 0 };
            }
            const trimmed = inputValue.trim();
            const opRegex = /^([\+\-\*\/])(\d+)$/;
            const match = opRegex.exec(trimmed);
            if (match) {
                return { op: match[1], num: parseInt(match[2]) || 0 };
            }
            return { op: '', num: 0 };
        }

        // 【核心】计算按钮组合的最终结果（无顺序，直接作用于初始值）
        function calculateCombinationResult(buttons) {
            let a = initA;
            let b = initB;

            // 遍历所有选中的按钮，运算直接作用于初始值（无顺序/优先级）
            buttons.forEach(btnIndex => {
                // A通道运算
                const aOpInput = document.querySelector(`.op-input[data-chan="A"][data-index="${btnIndex}"]`);
                const aOp = parseOp(aOpInput.value);
                switch (aOp.op) {
                    case '+': a += aOp.num; break;
                    case '-': a -= aOp.num; break;
                    case '*': a *= aOp.num; break;
                    case '/': a = aOp.num !== 0 ? Math.trunc(a / aOp.num) : a; break;
                }

                // B通道运算
                const bOpInput = document.querySelector(`.op-input[data-chan="B"][data-index="${btnIndex}"]`);
                const bOp = parseOp(bOpInput.value);
                switch (bOp.op) {
                    case '+': b += bOp.num; break;
                    case '-': b -= bOp.num; break;
                    case '*': b *= bOp.num; break;
                    case '/': b = bOp.num !== 0 ? Math.trunc(b / bOp.num) : b; break;
                }
            });

            return { a, b };
        }

        // 更新界面显示（仅保留结果计算）
        function updateUI() {
            // 计算并显示当前组合的结果
            const result = calculateCombinationResult(selectedButtons);
            resultA.textContent = result.a;
            resultB.textContent = result.b;
        }

        // 清空所有输入框内容（无弹窗提示）
        function clearAllInputs() {
            // 1. 清空目标值、初始值输入框
            document.getElementById('targetA').value = '';
            document.getElementById('targetB').value = '';
            document.getElementById('initA').value = '';
            document.getElementById('initB').value = '';

            // 2. 清空所有按钮运算规则输入框（A/B通道）
            const allOpInputs = document.querySelectorAll('.op-input');
            allOpInputs.forEach(input => {
                input.value = '';
            });

            // 3. 重置数值变量并更新结果显示
            initA = 0;
            initB = 0;
            targetA = 0;
            targetB = 0;
            resultA.textContent = '0';
            resultB.textContent = '0';
        }

        // 按钮选择/取消逻辑
        triggerBtns.forEach((btn, idx) => {
            btn.addEventListener('click', () => {
                if (selectedButtons.has(idx)) {
                    // 取消选择
                    selectedButtons.delete(idx);
                    btn.classList.remove('active');
                } else {
                    // 选择按钮
                    selectedButtons.add(idx);
                    btn.classList.add('active');
                }
                updateUI();
            });
        });

        // 重置选择
        resetBtn.addEventListener('click', () => {
            selectedButtons.clear();
            triggerBtns.forEach(btn => {
                btn.classList.remove('active', 'solution');
            });
            solutionTip.style.display = 'none';
            initValues();
            updateUI();
        });

        // 绑定清空数字事件
        clearInputsBtn.addEventListener('click', clearAllInputs);

        // 【核心】查找所有有效按钮组合（无顺序）
        function findValidCombinations() {
            initValues();
            triggerBtns.forEach(btn => btn.classList.remove('solution'));
            solutionTip.style.display = 'none';

            const allButtons = [0,1,2,3,4,5,6,7]; // 8个按钮的索引
            const validCombinations = [];

            // 生成所有可能的按钮组合（幂集），最多8个按钮
            function generateCombinations(arr, start, current) {
                // 计算当前组合的结果
                const result = calculateCombinationResult(new Set(current));
                if (result.a === targetA && result.b === targetB && current.length > 0) {
                    validCombinations.push([...current]);
                }

                // 递归生成组合（避免重复，按索引递增）
                for (let i = start; i < arr.length; i++) {
                    current.push(i);
                    generateCombinations(arr, i + 1, current);
                    current.pop();
                }
            }

            generateCombinations(allButtons, 0, []);

            // 展示结果
            if (validCombinations.length > 0) {
                // 优先展示最短的组合
                validCombinations.sort((a, b) => a.length - b.length);
                const bestCombination = validCombinations[0];
                const bestCombinationText = bestCombination.map(idx => `按钮${idx + 1}`).join('、');

                // 高亮有效组合的按钮
                bestCombination.forEach(idx => {
                    triggerBtns[idx].classList.add('solution');
                });

                // 仅保留组合信息和提示
                solutionTip.textContent = `✅ 找到有效组合（最少按钮）：${bestCombinationText}（游戏内任意顺序点击均有效）`;
                solutionTip.className = 'solution-tip success';
                solutionTip.style.display = 'block';

                // 自动选中最优组合
                selectedButtons = new Set(bestCombination);
                triggerBtns.forEach((btn, idx) => {
                    if (selectedButtons.has(idx)) {
                        btn.classList.add('active');
                    }
                });
                updateUI();
            } else {
                solutionTip.textContent = '❌ 未找到有效按钮组合，请检查初始值/目标值/按钮运算规则';
                solutionTip.className = 'solution-tip';
                solutionTip.style.display = 'block';
            }
        }

        // 绑定自动计算有效组合事件
        autoSolveBtn.addEventListener('click', findValidCombinations);

        // 监听数值输入变化
        document.getElementById('initA').addEventListener('input', () => {
            initValues();
            updateUI();
        });
        document.getElementById('initB').addEventListener('input', () => {
            initValues();
            updateUI();

            initValues();
            updateUI();
        });
        document.getElementById('targetB').addEventListener('input', () => {
            initValues();
            updateUI();
        });

        // 初始化
        initValues();
        updateUI();
